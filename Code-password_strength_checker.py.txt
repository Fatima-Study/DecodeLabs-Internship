print("======================================")
print("     PASSWORD STRENGTH CHECKER")
print("======================================")

password = input("Enter your password: ")

# Check password length
length_check = len(password) >= 8

# Check for uppercase letters
uppercase_check = any(char.isupper() for char in password)

# Check for numbers
number_check = any(char.isdigit() for char in password)

# Check for symbols
symbol_check = any(not char.isalnum() for char in password)

# Calculate score
score = sum([
    length_check,
    uppercase_check,
    number_check,
    symbol_check
])

# Determine password strength
if score <= 1:
    strength = "WEAK"
elif score == 2 or score == 3:
    strength = "MEDIUM"
else:
    strength = "STRONG"

print("\n========== RESULT ==========")
print("Password Strength:", strength)

print("\nSecurity Checks:")
print("Length (8+ characters):", "PASS" if length_check else "FAIL")
print("Uppercase Letter:", "PASS" if uppercase_check else "FAIL")
print("Number:", "PASS" if number_check else "FAIL")
print("Symbol:", "PASS" if symbol_check else "FAIL")

print("============================")
